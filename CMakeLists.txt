cmake_minimum_required(VERSION 3.24)
project(OblivionQuickLoot LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(FetchContent)
FetchContent_Declare(minhook
    GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
    GIT_TAG v1.3.4)
FetchContent_MakeAvailable(minhook)

add_library(OblivionQuickLoot SHARED
    src/dllmain.cpp
    src/scan.cpp
    src/game.cpp
    src/quickloot.cpp
    src/render.cpp
    src/draw.cpp
    src/assets.cpp
    src/gallery.cpp
    src/config.cpp
    src/log.cpp)
target_link_libraries(OblivionQuickLoot PRIVATE minhook)
target_compile_definitions(OblivionQuickLoot PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
# /EHa: SEH faults inside hooks unwind (releasing locks) into the catch(...) guard, which disables the mod.
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
target_compile_options(OblivionQuickLoot PRIVATE /W4 /permissive- /utf-8 /EHa)

# Loader for both Steam and Game Pass: a forwarding dsound.dll that loads OblivionQuickLoot.dll.
# proxy/dsound_exports.h is generated by proxy/gen_forwarders.py.
add_library(dsound_proxy SHARED proxy/dsound_proxy.cpp)
target_compile_definitions(dsound_proxy PRIVATE WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
set_target_properties(dsound_proxy PROPERTIES OUTPUT_NAME dsound)
